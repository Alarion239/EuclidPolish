% \fmap{name}{x}{y}{thickness}{height}{colour}: feature-map cuboid.
% Anchors: -w front west, -e back edge of the side face, -n top, -s bottom.
\newcommand{\fmap}[6]{%
  \pgfmathsetmacro{\fx}{#2}\pgfmathsetmacro{\fy}{#3}%
  \pgfmathsetmacro{\ft}{#4}\pgfmathsetmacro{\fh}{#5}%
  \pgfmathsetmacro{\fdx}{0.30*\fh}\pgfmathsetmacro{\fdy}{0.22*\fh}%
  \begin{scope}[line width=0.5pt,draw=#6!50!black,line join=round]
    \filldraw[fill=#6!22] (\fx,\fy+0.5*\fh) -- ++(\ft,0) -- ++(\fdx,\fdy) -- ++(-\ft,0) -- cycle;
    \filldraw[fill=#6!62] (\fx+\ft,\fy-0.5*\fh) -- ++(0,\fh) -- ++(\fdx,\fdy) -- ++(0,-\fh) -- cycle;
    \filldraw[fill=#6!42] (\fx,\fy-0.5*\fh) rectangle ++(\ft,\fh);
  \end{scope}
  \coordinate (#1-w) at (\fx,\fy);
  \coordinate (#1-e) at (\fx+\ft+\fdx,\fy);
  \coordinate (#1-n) at (\fx+0.5*\ft+0.5*\fdx,\fy+0.5*\fh+\fdy);
  \coordinate (#1-s) at (\fx+0.5*\ft,\fy-0.5*\fh);
}

% \slabs{x}{y}{thickness}{height}{n}{colour}: n equal slabs (the 8 knees)
\newcommand{\slabs}[6]{%
  \pgfmathsetmacro{\sx}{#1}\pgfmathsetmacro{\sy}{#2}%
  \pgfmathsetmacro{\st}{#3}\pgfmathsetmacro{\sh}{#4}%
  \pgfmathsetmacro{\sdx}{0.30*\sh}\pgfmathsetmacro{\sdy}{0.22*\sh}%
  \pgfmathtruncatemacro{\sn}{#5-1}%
  \foreach \i in {1,...,\sn}{%
    \pgfmathsetmacro{\xi}{\sx+\i*\st/#5}%
    \draw[#6!50!black,line width=0.35pt]
      (\xi,\sy-0.5*\sh) -- (\xi,\sy+0.5*\sh) -- ++(\sdx,\sdy);
  }%
}

% \imstack{name}{x}{y}{size}{image}: colour image in front of three planes
\newcommand{\imstack}[5]{%
  \pgfmathsetmacro{\bx}{#2}\pgfmathsetmacro{\by}{#3}\pgfmathsetmacro{\bs}{#4}%
  \foreach \k in {3,2,1}{%
    \filldraw[fill=white,draw=black!50,line width=0.4pt]
      ({\bx+0.07*\k},{\by-0.5*\bs+0.07*\k}) rectangle ++(\bs,\bs);
  }%
  \node[inner sep=0pt,anchor=south west] at (\bx,\by-0.5*\bs)
    {\includegraphics[width=\bs cm]{thumbs/#5.png}};
  \draw[black!80,line width=0.4pt] (\bx,\by-0.5*\bs) rectangle ++(\bs,\bs);
  \coordinate (#1-w) at (\bx,\by);
  \coordinate (#1-e) at (\bx+\bs+0.21,\by);
  \coordinate (#1-s) at (\bx+0.5*\bs,\by-0.5*\bs);
}
