% \fmap{name}{x}{y}{thickness = channels}{height = spatial}{colour}
% A feature-map cuboid: front face (thickness x height), oblique depth = width.
% Anchors: name-w (front west mid), name-e (side-face mid, same y),
%          name-n (top-face mid), name-s (front south mid).
\newcommand{\fmap}[6]{%
  \pgfmathsetmacro{\fx}{#2}\pgfmathsetmacro{\fy}{#3}%
  \pgfmathsetmacro{\ft}{#4}\pgfmathsetmacro{\fh}{#5}%
  \pgfmathsetmacro{\fdx}{0.30*\fh}\pgfmathsetmacro{\fdy}{0.22*\fh}%
  \begin{scope}[line width=0.4pt,draw=#6!55!black,line join=round]
    \filldraw[fill=#6!22] (\fx,\fy+0.5*\fh) -- ++(\ft,0) -- ++(\fdx,\fdy) -- ++(-\ft,0) -- cycle;
    \filldraw[fill=#6!60] (\fx+\ft,\fy-0.5*\fh) -- ++(0,\fh) -- ++(\fdx,\fdy) -- ++(0,-\fh) -- cycle;
    \filldraw[fill=#6!40] (\fx,\fy-0.5*\fh) rectangle ++(\ft,\fh);
  \end{scope}
  \coordinate (#1-w) at (\fx,\fy);
  \coordinate (#1-e) at (\fx+\ft+\fdx,\fy);
  \coordinate (#1-n) at (\fx+0.5*\ft+0.5*\fdx,\fy+0.5*\fh+\fdy);
  \coordinate (#1-s) at (\fx+0.5*\ft,\fy-0.5*\fh);
}

% \imstack{name}{x}{y}{size}{image}: a colour image in front of three blank
% planes (the four band planes of the cube)
\newcommand{\imstack}[5]{%
  \pgfmathsetmacro{\bx}{#2}\pgfmathsetmacro{\by}{#3}\pgfmathsetmacro{\bs}{#4}%
  \foreach \k in {3,2,1}{%
    \filldraw[fill=white,draw=black!45,line width=0.35pt]
      ({\bx+0.09*\k},{\by-0.5*\bs+0.09*\k}) rectangle ++(\bs,\bs);
  }%
  \node[inner sep=0pt,anchor=south west] at (\bx,\by-0.5*\bs)
    {\includegraphics[width=\bs cm]{thumbs/#5.png}};
  \draw[black!70,line width=0.35pt] (\bx,\by-0.5*\bs) rectangle ++(\bs,\bs);
  \coordinate (#1-w) at (\bx,\by);
  \coordinate (#1-e) at (\bx+\bs+0.27,\by);
  \coordinate (#1-s) at (\bx+0.5*\bs,\by-0.5*\bs);
  \coordinate (#1-n) at (\bx+0.5*\bs+0.135,\by+0.5*\bs+0.27);
}
